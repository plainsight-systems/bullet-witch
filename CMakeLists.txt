cmake_minimum_required(VERSION 3.25)
project(BulletWitchRebuild LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# Preserve the documented executable paths across module-local definitions.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
include(cmake/Quality.cmake)
include(cmake/Modules.cmake)
include(cmake/ThirdParty.cmake)
find_package(Threads REQUIRED)
include(CTest)

# Build policy only: no universal source include path.
add_library(bw_contracts INTERFACE)
target_compile_options(bw_contracts INTERFACE
  -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion)

add_subdirectory(src/platform)
add_subdirectory(src/engine/core/diagnostics)
add_subdirectory(src/engine/core/concurrency)
add_subdirectory(src/engine/core/tasks)
add_subdirectory(src/engine/core/memory)
add_subdirectory(src/engine/core/math)
add_subdirectory(src/engine/runtime)
add_subdirectory(src/engine/rendering/shaders)
add_subdirectory(src/engine/rendering/geometry)
add_subdirectory(src/engine/rendering/rhi)
add_subdirectory(src/engine/rendering/graph)
add_subdirectory(src/engine/rendering)
add_subdirectory(src/engine/rendering/materials)
add_subdirectory(src/engine/rendering/scene)
add_subdirectory(src/engine/animation)
add_subdirectory(src/engine/content)
add_subdirectory(src/hosts/macos)
add_subdirectory(tools/runtime)
add_subdirectory(tools/content_converter)
if(BW_SHADER_TOOLCHAIN)
  add_subdirectory(tools/shader_compiler)
  include(cmake/ShaderArtifacts.cmake)
endif()
add_subdirectory(tests)
include(cmake/HeaderChecks.cmake)
include(cmake/IncludeBoundaries.cmake)
